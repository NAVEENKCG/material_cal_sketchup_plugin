require 'sketchup.rb'
require 'extensions.rb'

module MaterialAreaCalculatorPro

  PLUGIN_ID      = 'material_area_calculator_pro'.freeze
  PLUGIN_VERSION = '2.0.0'.freeze

  # Unit conversion factors FROM square inches (SketchUp's internal unit)
  SQIN_TO_SQFT = 1.0 / 144.0
  SQIN_TO_SQM  = 0.00064516
  SQIN_TO_SQYD = 1.0 / 1296.0

  # ---------------------------------------------------------------------------
  # CORE: Recursive face collector with visibility filtering and transform-aware
  # area scaling. Traverses groups and component instances at any nesting depth.
  #
  # @param entities    [Sketchup::Entities]  entity collection to scan
  # @param transform   [Geom::Transformation] cumulative transform from parents
  # @param results     [Hash]                 accumulator: material_name => area_sqin
  # @param only_visible [Boolean]             skip hidden geometry and hidden layers
  # ---------------------------------------------------------------------------
  def self.collect_face_areas(entities, transform, results, only_visible)
    entities.each do |entity|

      # ── Visibility gate ────────────────────────────────────────────────────
      if only_visible
        next if entity.hidden?
        # layer/tag visibility (method name is .layer in all SketchUp versions)
        next if entity.respond_to?(:layer) && entity.layer && !entity.layer.visible?
      end

      case entity

      # ── Face ──────────────────────────────────────────────────────────────
      when Sketchup::Face
        # Apply the cumulative parent transform to get the true world-space area.
        # Geom::Transformation#determinant gives the volume-scale factor; its
        # square-root gives the area-scale factor.
        scale_factor = Math.sqrt(transform.determinant.abs)
        area_sqin    = entity.area * scale_factor

        # FIX #2 — single-side counting
        # We attribute the area to whichever material is on the FRONT face.
        # If the front has no material, fall back to the back face material.
        # This prevents a face painted "Red" on both sides from being counted twice.
        mat = entity.material || entity.back_material
        next if mat.nil?

        results[mat.name] = (results[mat.name] || 0.0) + area_sqin

      # ── Group ─────────────────────────────────────────────────────────────
      when Sketchup::Group
        child_transform = transform * entity.transformation
        collect_face_areas(entity.entities, child_transform, results, only_visible)

      # ── Component instance ────────────────────────────────────────────────
      when Sketchup::ComponentInstance
        child_transform = transform * entity.transformation
        collect_face_areas(entity.definition.entities, child_transform, results, only_visible)

      end
    end
  end

  # ---------------------------------------------------------------------------
  # Run a full model scan and return a hash of { material_name => area_sqin }
  # ---------------------------------------------------------------------------
  def self.scan_model(only_visible: true)
    model   = Sketchup.active_model
    results = {}
    identity = Geom::Transformation.new   # identity transform — no scaling yet
    collect_face_areas(model.entities, identity, results, only_visible)
    results
  end

  # ---------------------------------------------------------------------------
  # Format a square-inch value into all four unit strings
  # ---------------------------------------------------------------------------
  def self.format_row(sq_in)
    {
      sqft: (sq_in * SQIN_TO_SQFT).round(2),
      sqm:  (sq_in * SQIN_TO_SQM ).round(4),
      sqyd: (sq_in * SQIN_TO_SQYD).round(3),
      sqin: sq_in.round(1)
    }
  end

  # ---------------------------------------------------------------------------
  # Build JSON payload for the HTML dialog (sorted by area descending)
  # ---------------------------------------------------------------------------
  def self.build_json(results)
    rows = results
      .sort_by { |_, v| -v }
      .map do |name, sq_in|
        r = format_row(sq_in)
        # Escape quotes in material names for JSON safety
        safe_name = name.gsub('\\', '\\\\').gsub('"', '\\"')
        "{\"name\":\"#{safe_name}\"," \
        "\"sqft\":#{r[:sqft]}," \
        "\"sqm\":#{r[:sqm]}," \
        "\"sqyd\":#{r[:sqyd]}," \
        "\"sqin\":#{r[:sqin]}}"
      end
    "[#{rows.join(',')}]"
  end

  # ---------------------------------------------------------------------------
  # FIX #4 — Export with UI.savepanel so the user picks filename & location
  # ---------------------------------------------------------------------------
  def self.export_csv(results)
    # FIX #4: open a native Save As dialog instead of silently overwriting
    path = UI.savepanel('Save material areas as CSV', Dir.home, 'material_areas.csv')
    return if path.nil? # user cancelled

    # Ensure .csv extension
    path += '.csv' unless path.end_with?('.csv')

    begin
      File.open(path, 'w') do |f|
        f.puts 'Material Name,Area (sq ft),Area (sq m),Area (sq yd),Area (sq in)'
        results.sort_by { |_, v| -v }.each do |name, sq_in|
          r = format_row(sq_in)
          f.puts "\"#{name.gsub('"', '""')}\",#{r[:sqft]},#{r[:sqm]},#{r[:sqyd]},#{r[:sqin]}"
        end
      end
      UI.messagebox("Exported successfully to:\n#{path}")
    rescue => e
      UI.messagebox("Export failed:\n#{e.message}")
    end
  end

            filtered = ALL_DATA.filter(function(r) {
              return r.name.toLowerCase().indexOf(q) !== -1;

            renderTable();

            if (sortCol === col) { sortAsc = !sortAsc; }
            else { sortCol = col; sortAsc = col === 'name'; }
            sortData();
            renderTable();
          }
          function sortData() {
            filtered.sort(function(a, b) {
              var av = a[sortCol], bv = b[sortCol];
              if (typeof av === 'string') {
                return sortAsc ? av.localeCompare(bv) : bv.localeCompare(av);
              }
              return sortAsc ? av - bv : bv - av;
            });
          }

            return s.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');

  # ---------------------------------------------------------------------------
  # Show the main HtmlDialog window
  # ---------------------------------------------------------------------------
  def self.show_dialog(only_visible: true)
    results = scan_model(only_visible: only_visible)

    if results.empty?
      UI.messagebox(
        "No materials with painted faces found.\n\n" \
        "#{only_visible ? 'Note: hidden geometry is excluded. Try toggling the visibility filter.' : ''}"
      )
      return
    end

    json = build_json(results)

    dlg = UI::HtmlDialog.new(
      dialog_title:    'Material Area Calculator Pro',
      preferences_key: PLUGIN_ID,
      width:  640,
      height: 480,
      resizable: true
    )

    dlg.set_html(dialog_html(json, only_visible))

    # ── FIX #1: Modern async JS bridge callbacks ──────────────────────────
    # export_csv — open native Save As dialog
    dlg.add_action_callback('export_csv') do |_ctx|
      export_csv(results)
    end

    # rescan — close this dialog, re-scan, reopen
    dlg.add_action_callback('rescan') do |_ctx|
      dlg.close
      show_dialog(only_visible: only_visible)
    end

    dlg.show
  end
